set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
cmake_minimum_required(VERSION 3.0)
project(hz_page_art LANGUAGES C)
set(CMAKE_C_STANDARD 11)
find_package(Freetype REQUIRED)
find_package(PNG REQUIRED)
find_package(Threads REQUIRED)
get_filename_component(UI_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(VEHICLE_SOURCE "${UI_ROOT}/../../utils/vehicle/src" CACHE PATH "Vehicle API headers")
file(GLOB_RECURSE LVGL "${UI_ROOT}/src/hmi/lvgl/src/*.c")
file(GLOB_RECURSE DASH "${UI_ROOT}/src/dashboard/UI/*.c" "${UI_ROOT}/src/dashboard/obj/*.c")
set(HOST navigation page_header settings settings_store gear_calibration gear_calibration_store static_scene transition page_art art_codec system_fonts vehicle_ui vehicle_body pdc_view rk_offline preview_guard music_player video_player video_ui bt_music android_auto)
set(SOURCES)
foreach(SOURCE ${HOST})
 list(APPEND SOURCES "${UI_ROOT}/host/${SOURCE}.c")
endforeach()
add_executable(hz-page-art main.c pdc.c ${LVGL} ${DASH} ${SOURCES} "${VEHICLE_SOURCE}/vehicle_calibration.c"
 "${UI_ROOT}/src/hmi/mode/Car_Launcher.c" "${UI_ROOT}/src/hmi/mode/Car_Dashboard.c")
target_include_directories(hz-page-art PRIVATE "${UI_ROOT}/src/hmi" "${UI_ROOT}/src/hmi/lvgl" "${UI_ROOT}/src/dashboard" "${UI_ROOT}/host" "${VEHICLE_SOURCE}" ${FREETYPE_INCLUDE_DIRS} ${PNG_INCLUDE_DIRS})
target_compile_definitions(hz-page-art PRIVATE HZ_TARGET_UI HZ_COMBINED_UI LV_CONF_INCLUDE_SIMPLE LV_LVGL_H_INCLUDE_SIMPLE)
target_compile_definitions(hz-page-art PRIVATE HZ_PDC_SOURCE="${UI_ROOT}/assets/pdc-car.png")
target_compile_definitions(hz-page-art PRIVATE HZ_VEHICLE_SOURCE="${UI_ROOT}/assets/vehicle")
target_compile_definitions(hz-page-art PRIVATE HZ_LAUNCHER_SOURCE="${UI_ROOT}/assets/launcher")
target_compile_options(hz-page-art PRIVATE -O2 -ffunction-sections -fdata-sections)
set_target_properties(hz-page-art PROPERTIES LINK_FLAGS "-Wl,--gc-sections -Wl,--wrap=system -Wl,--wrap=fopen")
target_link_libraries(hz-page-art PRIVATE ${PNG_LIBRARIES} ${FREETYPE_LIBRARIES} ${CMAKE_THREAD_LIBS_INIT} m)
