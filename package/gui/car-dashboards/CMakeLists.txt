cmake_minimum_required(VERSION 3.16)
project(dashboard_preview LANGUAGES C)
set(CMAKE_C_STANDARD 11)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
pkg_check_modules(FONTCONFIG REQUIRED IMPORTED_TARGET fontconfig)
find_package(Threads REQUIRED)
set(VEHICLE_PREFIX "${CMAKE_CURRENT_SOURCE_DIR}/../../../out/vehicle/host/prefix" CACHE PATH "Local ubus host dependency prefix")
set(VEHICLE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../utils/vehicle/src")
find_path(VEHICLE_UBUS_INCLUDE libubus.h HINTS "${VEHICLE_PREFIX}/include")
find_library(VEHICLE_UBUS_LIBRARY ubus HINTS "${VEHICLE_PREFIX}/lib")
find_library(VEHICLE_UBOX_LIBRARY ubox HINTS "${VEHICLE_PREFIX}/lib")
if(VEHICLE_UBUS_INCLUDE AND VEHICLE_UBUS_LIBRARY AND VEHICLE_UBOX_LIBRARY)
    add_library(preview_vehicle STATIC "${VEHICLE_SOURCE}/vehicle.c"
        "${VEHICLE_SOURCE}/vehicle_client.c" "${VEHICLE_SOURCE}/vehicle_blob.c" "${VEHICLE_SOURCE}/vehicle_calibration.c")
    target_include_directories(preview_vehicle PUBLIC "${VEHICLE_SOURCE}" "${VEHICLE_UBUS_INCLUDE}")
    target_link_libraries(preview_vehicle PUBLIC "${VEHICLE_UBUS_LIBRARY}" "${VEHICLE_UBOX_LIBRARY}")
    target_compile_definitions(preview_vehicle PUBLIC HAVE_VEHICLE_UBUS)
endif()
if(NOT DEFINED PREVIEW_SOURCE)
    set(PREVIEW_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/src/${PREVIEW_PROJECT}")
endif()
if(NOT IS_DIRECTORY "${PREVIEW_SOURCE}/lvgl/src")
    message(FATAL_ERROR "PREVIEW_SOURCE must contain a populated LVGL tree")
endif()
file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS
    "${PREVIEW_SOURCE}/lvgl/src/*.c")
add_library(preview_lvgl STATIC ${LVGL_SOURCES} host/system_fonts.c)
target_include_directories(preview_lvgl PUBLIC
    "${PREVIEW_SOURCE}" "${PREVIEW_SOURCE}/lvgl" "${CMAKE_CURRENT_SOURCE_DIR}/host")
target_compile_definitions(preview_lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE LV_LVGL_H_INCLUDE_SIMPLE)
target_link_libraries(preview_lvgl PUBLIC PkgConfig::SDL2 PkgConfig::FONTCONFIG Threads::Threads m)
target_compile_definitions(preview_lvgl PUBLIC LV_DRV_CONF_INCLUDE_SIMPLE)
target_compile_options(preview_lvgl PRIVATE -ffunction-sections -fdata-sections)
if(PREVIEW_PROJECT STREQUAL "dashboard")
    find_package(Freetype REQUIRED)
    target_link_libraries(preview_lvgl PUBLIC Freetype::Freetype)
    target_compile_definitions(preview_lvgl PUBLIC LV_DRV_CONF_INCLUDE_SIMPLE)
    file(GLOB_RECURSE APP_SOURCES CONFIGURE_DEPENDS
        "${PREVIEW_SOURCE}/UI/*.c" "${PREVIEW_SOURCE}/obj/*.c"
        "${PREVIEW_SOURCE}/lv_drivers/sdl/*.c")
    list(APPEND APP_SOURCES host/v8_main.c)
elseif(PREVIEW_PROJECT STREQUAL "hmi")
    find_package(Freetype REQUIRED)
    target_link_libraries(preview_lvgl PUBLIC Freetype::Freetype)
    set(APP_SOURCES "${PREVIEW_SOURCE}/mode/Car_Launcher.c"
        "${PREVIEW_SOURCE}/mode/Car_Dashboard.c"
        host/v8_main.c host/rk_offline.c host/music_player.c host/video_player.c host/video_ui.c host/bt_music.c host/android_auto.c)
    file(GLOB_RECURSE DASHBOARD_SOURCES CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/src/dashboard/UI/*.c"
        "${CMAKE_CURRENT_SOURCE_DIR}/src/dashboard/obj/*.c")
    list(APPEND APP_SOURCES ${DASHBOARD_SOURCES} host/navigation.c host/vehicle_body.c host/page_header.c host/settings.c host/settings_store.c host/gear_calibration.c host/gear_calibration_store.c host/static_scene.c host/transition.c host/page_art.c host/art_codec.c)
    if(NOT TARGET preview_vehicle)
        list(APPEND APP_SOURCES "${VEHICLE_SOURCE}/vehicle_calibration.c")
    endif()
else()
    message(FATAL_ERROR "Unknown preview: ${PREVIEW_PROJECT}")
endif()
if(NOT PREVIEW_PROJECT STREQUAL "dashboard")
    file(GLOB SDL_SOURCES "${PREVIEW_SOURCE}/lv_drivers/sdl/*.c")
    list(APPEND APP_SOURCES ${SDL_SOURCES})
endif()
add_executable(preview ${APP_SOURCES} host/preview_guard.c host/vehicle_ui.c host/pdc_view.c host/art_codec.c)
target_include_directories(preview PRIVATE "${VEHICLE_SOURCE}")
if(TARGET preview_vehicle)
    target_link_libraries(preview PRIVATE preview_vehicle)
endif()
target_compile_options(preview PRIVATE -ffunction-sections -fdata-sections)
target_include_directories(preview PRIVATE "${PREVIEW_SOURCE}/UI" "${PREVIEW_SOURCE}/lv_drivers")
target_link_libraries(preview PRIVATE preview_lvgl)
target_link_options(preview PRIVATE
    -Wl,--wrap=system -Wl,--wrap=fopen -Wl,--gc-sections)
if(PREVIEW_PROJECT STREQUAL "dashboard")
    target_compile_definitions(preview PRIVATE PREVIEW_LVGL_DASHBOARD)
endif()
if(PREVIEW_PROJECT STREQUAL "hmi")
    target_compile_definitions(preview PRIVATE HZ_COMBINED_UI)
    target_include_directories(preview BEFORE PRIVATE "${PREVIEW_SOURCE}" "${CMAKE_CURRENT_SOURCE_DIR}/src/dashboard")
    add_executable(pdc-preview host/v8_main.c host/vehicle_ui.c host/pdc_view.c host/art_codec.c ${SDL_SOURCES})
    target_compile_definitions(pdc-preview PRIVATE PREVIEW_PDC_APP)
    target_include_directories(pdc-preview PRIVATE "${PREVIEW_SOURCE}/lv_drivers" "${VEHICLE_SOURCE}")
    target_link_libraries(pdc-preview PRIVATE preview_lvgl)
    if(TARGET preview_vehicle)
        target_link_libraries(pdc-preview PRIVATE preview_vehicle)
        # Exercise the production memory/font configuration without fbdev.
        add_library(target_ui_widgets STATIC ${LVGL_SOURCES} host/system_fonts.c
            "${PREVIEW_SOURCE}/mode/Car_Launcher.c" "${PREVIEW_SOURCE}/mode/Car_Dashboard.c"
            ${DASHBOARD_SOURCES} host/navigation.c host/vehicle_body.c host/page_header.c host/settings.c host/settings_store.c host/gear_calibration.c host/gear_calibration_store.c host/static_scene.c host/transition.c host/page_art.c host/art_codec.c host/rk_offline.c
            host/preview_guard.c host/vehicle_ui.c host/pdc_view.c host/music_player.c host/video_player.c host/video_ui.c host/bt_music.c host/android_auto.c)
        target_compile_definitions(target_ui_widgets PUBLIC HZ_TARGET_UI HZ_COMBINED_UI
            HAVE_VEHICLE_UBUS LV_CONF_INCLUDE_SIMPLE LV_LVGL_H_INCLUDE_SIMPLE)
        target_include_directories(target_ui_widgets PUBLIC "${PREVIEW_SOURCE}" "${PREVIEW_SOURCE}/lvgl"
            "${CMAKE_CURRENT_SOURCE_DIR}/src/dashboard" "${CMAKE_CURRENT_SOURCE_DIR}/host" "${VEHICLE_SOURCE}")
        target_compile_options(target_ui_widgets PRIVATE -ffunction-sections -fdata-sections)
        target_link_libraries(target_ui_widgets PUBLIC Freetype::Freetype Threads::Threads preview_vehicle m)
        add_executable(hz-ui-check "${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/target-ui/ui.c")
        target_compile_options(hz-ui-check PRIVATE -UNDEBUG)
        target_link_libraries(hz-ui-check PRIVATE target_ui_widgets)
        target_link_options(hz-ui-check PRIVATE -Wl,--gc-sections -Wl,--wrap=system -Wl,--wrap=fopen -Wl,--wrap=time)
        add_executable(hz-gear-calibration-check "${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/target-ui/gear-calibration.c")
        target_compile_options(hz-gear-calibration-check PRIVATE -UNDEBUG -Wall -Wextra -Werror)
        target_link_libraries(hz-gear-calibration-check PRIVATE target_ui_widgets)
        target_link_options(hz-gear-calibration-check PRIVATE -Wl,--gc-sections -Wl,--wrap=system -Wl,--wrap=fopen)
        add_executable(hz-gradient-check "${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/target-ui/gradient.c")
        target_compile_options(hz-gradient-check PRIVATE -UNDEBUG -Wall -Wextra -Werror)
        target_link_libraries(hz-gradient-check PRIVATE target_ui_widgets)
        add_executable(hz-g2d-check target/g2d_display.c target/linux_display.c host/static_scene.c host/transition.c
            "${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/target-ui/g2d.c")
        target_compile_definitions(hz-g2d-check PRIVATE HZ_USE_G2D)
        target_include_directories(hz-g2d-check PRIVATE target
            "${CMAKE_CURRENT_SOURCE_DIR}/../littlevgl-8/lv_drivers/display"
            "${CMAKE_CURRENT_SOURCE_DIR}/../../allwinner/libuapi/src/include")
        target_compile_options(hz-g2d-check PRIVATE -UNDEBUG -Wall -Wextra -Werror)
        target_link_libraries(hz-g2d-check PRIVATE target_ui_widgets)
        target_link_options(hz-g2d-check PRIVATE -Wl,--gc-sections
            -Wl,--wrap=open -Wl,--wrap=close -Wl,--wrap=ioctl -Wl,--wrap=mmap
            -Wl,--wrap=munmap -Wl,--wrap=read)
    endif()
endif()
