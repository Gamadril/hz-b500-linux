cmake_minimum_required(VERSION 3.16)
project(vehicle C)
set(CMAKE_C_STANDARD 11)
add_compile_options(-Wall -Wextra -Wformat=2)
find_path(UBUS_INCLUDE_DIR libubus.h REQUIRED)
find_library(UBUS_LIBRARY ubus REQUIRED)
find_library(UBOX_LIBRARY ubox REQUIRED)
find_library(BLOBMSG_JSON_LIBRARY blobmsg_json REQUIRED)
add_library(vehicle-core STATIC src/vehicle.c src/vehicle_calibration.c)
target_include_directories(vehicle-core PUBLIC src)
add_library(vehicle-client STATIC src/vehicle_client.c src/vehicle_blob.c)
target_include_directories(vehicle-client PUBLIC src)
target_include_directories(vehicle-client SYSTEM PUBLIC ${UBUS_INCLUDE_DIR})
target_link_libraries(vehicle-client PUBLIC vehicle-core ${UBUS_LIBRARY} ${UBOX_LIBRARY})
add_executable(vehicled src/vehicled.c src/vehicle_clock.c)
target_link_libraries(vehicled PRIVATE vehicle-client)
add_executable(vehicle-watch src/vehicle_watch.c)
target_link_libraries(vehicle-watch PRIVATE vehicle-client ${BLOBMSG_JSON_LIBRARY})
add_executable(vehicle-slcan src/vehicle_slcan.c)
target_link_libraries(vehicle-slcan PRIVATE vehicle-core)
install(TARGETS vehicled vehicle-watch vehicle-slcan RUNTIME DESTINATION bin)
install(FILES src/vehicle.h src/vehicle_client.h src/vehicle_calibration.h DESTINATION include/vehicle)
install(TARGETS vehicle-core vehicle-client ARCHIVE DESTINATION lib)
